            </button>

        </header>


        <!-- LLISTA DE PLANTES -->

            <!-- Es carregarà automàticament -->

        </div>

    </div>


    <!-- MODAL PER AFEGIR PLANTA -->

    <div id="modal" class="hidden fixed inset-0 bg-black bg-opacity-50 flex items-center justify-center p-4">

        <div class="bg-white p-6 rounded-xl w-full max-w-sm">

            <h2 class="text-xl font-bold mb-4">Nova Planta</h2>

            <input type="text" id="plantName" placeholder="Nom de la planta" class="w-full border p-2 mb-3 rounded">

            <label class="text-sm text-gray-600">Cada quants dies es rega?</label>

            <input type="number" id="plantFreq" value="3" class="w-full border p-2 mb-3 rounded">

            <label class="text-sm text-gray-600">Foto de la planta:</label>

            <input type="file" id="plantPhoto" accept="image/*" class="w-full mb-4 text-sm">

            

            <div class="flex justify-end space-x-2">

                <button onclick="document.getElementById('modal').classList.add('hidden')" class="px-4 py-2 text-gray-500">Cancel·lar</button>

                <button onclick="addPlant()" class="px-4 py-2 bg-green-600 text-white rounded">Guardar</button>

            </div>

        </div>

    </div>


    <script>

        let plants = JSON.parse(localStorage.getItem('myPlants')) || [];


        function addPlant() {

            const name = document.getElementById('plantName').value;

            const freq = document.getElementById('plantFreq').value;

            const photoFile = document.getElementById('plantPhoto').files[0];


            if (!name || !photoFile) return alert("Posa un nom i una foto!");


            const reader = new FileReader();

            reader.onloadend = function() {

                const newPlant = {

                    id: Date.now(),

                    name: name,

                    freq: parseInt(freq),

                    lastWatered: new Date().toISOString(),

                    photo: reader.result

                };

                plants.push(newPlant);

                saveAndRender();

                document.getElementById('modal').classList.add('hidden');

            }

            reader.readAsDataURL(photoFile);

        }


        function waterPlant(id) {

            plants = plants.map(p => {

                if (p.id === id) p.lastWatered = new Date().toISOString();

                return p;

            });

            saveAndRender();

        }


        function deletePlant(id) {

            if (confirm("Segur que vols eliminar-la?")) {

                plants = plants.filter(p => p.id !== id);

                saveAndRender();

            }

        }


        function saveAndRender() {

            localStorage.setItem('myPlants', JSON.stringify(plants));

            renderPlants();

        }


        function renderPlants() {

            const list = document.getElementById('plantList');

            list.innerHTML = "";


            plants.forEach(p => {

                const lastDate = new Date(p.lastWatered);

                const nextDate = new Date(lastDate);

                nextDate.setDate(lastDate.getDate() + p.freq);

                

                const avui = new Date();

                const isDue = avui >= nextDate;


                const card = document.createElement('div');

                card.className = "bg-white rounded-xl shadow-md overflow-hidden flex items-center p-3 border-l-4 " + (isDue ? "border-red-500" : "border-green-500");

                

                card.innerHTML = `

                    <img src="${p.photo}" class="w-20 h-20 object-cover rounded-lg mr-4">

                    <div class="flex-1">

                        <h3 class="font-bold text-gray-800">${p.name}</h3>

                        <p class="text-xs text-gray-500">Regat: ${lastDate.toLocaleDateString()}</p>

                        <p class="text-sm ${isDue ? 'due-date' : 'text-green-600'}">

                            Proper reg: ${nextDate.toLocaleDateString()}

                        </p>

                    </div>

                    <div class="flex flex-col space-y-2">

                        <button onclick="waterPlant(${p.id})" class="bg-blue-100 text-blue-600 p-2 rounded text-xs font-bold text-center">💧 REGAR</button>

                        <button onclick="deletePlant(${p.id})" class="text-gray-300 text-xs text-right">Eliminar</button>

                    </div>

                `;

                list.appendChild(card);

            });

        }


        renderPlants();

    </script>
